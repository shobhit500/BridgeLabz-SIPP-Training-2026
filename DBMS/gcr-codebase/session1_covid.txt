
-- Challenge 1: Create COVID Database Schema

-- Task A: Create Database
CREATE DATABASE covid_db;

-- Select Database
USE covid_db;

-- Task B: Create covid_cases Table
CREATE TABLE covid_cases (
    id INT AUTO_INCREMENT PRIMARY KEY,
    date DATE,
    time VARCHAR(20),
    state VARCHAR(100),
    confirmed INT,
    deaths INT,
    recovered INT
);

-- Output:
-- Query OK, 1 row affected
-- Database changed
-- Query OK, 0 rows affected

-- Challenge 2: Insert Initial COVID Data

INSERT INTO covid_cases
(date, time, state, confirmed, deaths, recovered)
VALUES
('2020-03-15', '10 AM', 'India', 100, 2, 50);

-- Verify
SELECT * FROM covid_cases;
