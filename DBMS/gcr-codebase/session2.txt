-- Use Case 1: Define Primary Keys

ALTER TABLE covid_cases
ADD PRIMARY KEY (Country, Date);

-- Output:
Query OK, 0 rows affected

-- Verify

+-----------------+--------------+------+-----+---------+-------+
| Field           | Type         | Null | Key | Default | Extra |
+-----------------+--------------+------+-----+---------+-------+
| Country         | varchar(100) | NO   | PRI | NULL    |       |
| Date            | date         | NO   | PRI | NULL    |       |
| Confirmed_Cases | int          | YES  |     | NULL    |       |
| Deaths          | int          | YES  |     | NULL    |       |
| Recoveries      | int          | YES  |     | NULL    |       |
+-----------------+--------------+------+-----+---------+-------+


-- Use Case 2: Create Foreign Key Relationships

CREATE TABLE countries (
    Country VARCHAR(100) PRIMARY KEY,
    Continent VARCHAR(100)
);

ALTER TABLE covid_cases
ADD CONSTRAINT fk_country
FOREIGN KEY (Country)
REFERENCES countries(Country);

-- Output:
Query OK, 0 rows affected

-- Verify
SHOW CREATE TABLE covid_cases;

+-------------+------------------------------------------------------------------+
| Table       | Create Table                                                     |
+-------------+------------------------------------------------------------------+
| covid_cases | FOREIGN KEY (Country) REFERENCES countries(Country)              |
+-------------+------------------------------------------------------------------+


-- Use Case 3: Add Constraints to Ensure Data Quality

ALTER TABLE covid_cases
MODIFY Country VARCHAR(100) NOT NULL,
MODIFY Date DATE NOT NULL,
MODIFY Confirmed_Cases INT NOT NULL;

ALTER TABLE covid_cases
ADD CONSTRAINT chk_deaths
CHECK (Deaths <= Confirmed_Cases);

-- Output:
Query OK, 0 rows affected

-- Verify
SHOW CREATE TABLE covid_cases;

+-------------+--------------------------------------------------------------+
| Table       | Create Table                                                 |
+-------------+--------------------------------------------------------------+
| covid_cases | CHECK (Deaths <= Confirmed_Cases)                            |
+-------------+--------------------------------------------------------------+


-- Use Case 4: Set Default Values

-- Use Case 5: Prevent Duplicate Country-Date Entries

ALTER TABLE covid_cases
